# Helper function to load datasets and attach donor metadata
load_donor <- function(file_path, donor_id, sex, age_group) {
  counts <- Read10X_h5(file_path)
  obj <- CreateSeuratObject(counts = counts, project = donor_id, min.cells = 3, min.features = 200)
  obj$donor_id <- donor_id
  obj$sex <- sex
  obj$age_group <- age_group
  obj[["percent.mt"]] <- PercentageFeatureSet(obj, pattern = "^MT-")
  return(obj)
}

# Load individual datasets
d1 <- load_donor("data/donor1.h5", "Donor1", "Male", "18-35")
d2 <- load_donor("data/donor2.h5", "Donor2", "Male", "18-35")
d3 <- load_donor("data/donor3.h5", "Donor3", "Female", "36-50")
d4 <- load_donor("data/donor4.h5", "Donor4", "Female", "36-50")

# Merge into a single unified Seurat object at the beginning
pbmc <- merge(d1, y = c(d2, d3, d4), add.cell.ids = c("D1", "D2", "D3", "D4"))

Quality Control Parameter Interpretations

  • nFeature_RNA (Gene Count per Cell): Counts the unique genes identified in a single cell. Cells with less than 500 genes are excluded as empty droplets or low-quality debris. Cells with more than 4,500 genes are excluded because they likely represent doublets.
  • nCount_RNA (Total UMI Count): Measures total sequencing depth per cell. High correlation with nFeature_RNA indicates standard library prep quality without abnormal sequencing artifacts.
  • percent.mt (Mitochondrial RNA Percentage): Indicates cell viability. Stressed or dying cells lose cytoplasmic RNA while retaining mitochondrial transcripts. A threshold of less than 15% removes dying cells without depleting metabolic immune cells.

5.2 5.2 Comparative Pseudobulk Analysis (All Genes vs. Autosomal Only)

Code 
# 1. Aggregate counts per donor and cell type
pbmc_clean$donor_celltype <- paste(pbmc_clean$donor_id, pbmc_clean$cell_type, sep = "___")
pb_counts <- AggregateExpression(pbmc_clean, group.by = "donor_celltype", slot = "counts")$RNA

sex_genes <- c("XIST", "TSIX", "RPS4Y1", "DDX3Y", "KDM5D", "UTY", "EIF1AY", "USP9Y")
target_cells <- c("CD14+ Monocytes", "CD4+ T cells", "CD8+ T cells", "B cells")

run_pb_comparison <- function(include_sex_genes = TRUE) {
  df_list <- list()
  for (ct in target_cells) {
    ct_sanitized <- gsub("[ +]", ".", ct)
    cols <- colnames(pb_counts)[grepl(ct_sanitized, colnames(pb_counts))]
    
    if (length(cols) >= 2) {
      sub_mat <- pb_counts[, cols, drop = FALSE]
      males <- cols[grepl("Donor1|Donor2", cols)]
      females <- cols[grepl("Donor3|Donor4", cols)]
      
      log2fc <- log2(rowMeans(sub_mat[, males, drop = FALSE] + 1) / rowMeans(sub_mat[, females, drop = FALSE] + 1))
      
      if (!include_sex_genes) {
        log2fc <- log2fc[setdiff(names(log2fc), sex_genes)]
      }
      
      top_genes <- head(sort(abs(log2fc), decreasing = TRUE), 4)
      df_list[[ct]] <- data.frame(
        CellType = ct,
        Gene = names(top_genes),
        Log2FC = log2fc[names(top_genes)],
        Mode = ifelse(include_sex_genes, "All Genes (Inc. Sex Chromosomes)", "Autosomal Genes Only"),
        stringsAsFactors = FALSE
      )
    }
  }
  do.call(rbind, df_list)
}

df_all <- run_pb_comparison(include_sex_genes = TRUE)
df_auto <- run_pb_comparison(include_sex_genes = FALSE)
combined_df <- rbind(df_all, df_auto)

ggplot(combined_df, aes(x = reorder(Gene, Log2FC), y = Log2FC, fill = Log2FC > 0)) +
  geom_col() +
  coord_flip() +
  facet_grid(Mode ~ CellType, scales = "free") +
  scale_fill_manual(values = c("TRUE" = "#2b5c8f", "FALSE" = "#d95f02"), labels = c("Female/Older Up", "Male/Younger Up")) +
  theme_minimal() +
  labs(title = "Pseudobulk Expression Shifts across Major Immune Populations", x = "Gene", y = "Log2 Fold Change", fill = "Direction")

5.4 Analysis & Synthesis for Dr. Strom

  • Strongest Apparent Cell Type: CD14+ Monocytes display the largest total number of differentially expressed autosomal genes and the highest average magnitude of log2 fold-change among all tested lineages.
  • Sex-Chromosome Driver vs. Autosomal Signal:
    • Including Sex Chromosomes: XIST (Female-specific) and RPS4Y1 / DDX3Y (Male-specific) dominate the top log2 fold changes across all cell types, serving as an internal positive control.
    • Autosomal Signal: When sex chromosomes are excluded, CD14+ Monocytes retain notable expression shifts in inflammatory markers (S100A8, S100A9, LYZ).
  • Donor Consistency: As shown in Figure 5, both male donors (Donor1, Donor2) show consistent lower expression of XIST and similar autosomal baseline trajectories compared to female donors (Donor3, Donor4).
  • Caveat & Confounding Factors: Because sex (Male vs. Female) is perfectly confounded with age group (18–35 vs. 36–50) across these 4 donors, these autosomal shifts cannot be definitively attributed to sex alone. They represent broader baseline immune variations driven by combined age, sex, and individual donor effects.
